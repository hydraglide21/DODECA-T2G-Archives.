# ---------------- F2[X] ----------------
def degF(p): return p.bit_length()-1 if p else -1
def mulF(a,b):
    r=0
    while b:
        if b&1: r^=a
        a<<=1; b>>=1
    return r
def divmodF(a,b):
    assert b!=0
    q,r,db=0,a,degF(b)
    while r and degF(r)>=db:
        s=degF(r)-db; q^=(1<<s); r^=(b<<s)
    return q,r
def modF(a,b): return divmodF(a,b)[1]
def gcdF(a,b):
    while b: a,b=b,modF(a,b)
    return a
def derivF(p):
    r=0
    for i in range(degF(p)+1):
        if ((p>>i)&1) and (i%2==1): r^=(1<<(i-1))
    return r
def pow_x_2n_mod(p,n):
    r=0b10
    for _ in range(n): r=modF(mulF(r,r),p)
    return r
def prime_divisors(n):
    res,d,m=[],2,n
    while d*d<=m:
        if m%d==0:
            res.append(d)
            while m%d==0: m//=d
        d+=1 if d==2 else 2
    if m>1: res.append(m)
    return res
def is_irred_rabin(p):
    n=degF(p)
    if n<=0: return False
    if ((p>>n)&1)!=1: return False
    if n==1: return True
    if (p&1)==0: return False
    exp=modF(0b10,p)
    if pow_x_2n_mod(p,n)!=exp: return False
    for ell in prime_divisors(n):
        e=n//ell
        if gcdF(pow_x_2n_mod(p,e)^exp,p)!=1: return False
    return True
def factor_F2(p):
    assert p!=0
    if p==1: return {}
    facs,rem={},p
    while rem!=1 and (rem&1)==0:
        assert modF(rem,0b10)==0
        facs[0b10]=facs.get(0b10,0)+1
        rem=divmodF(rem,0b10)[0]
    d=1
    while rem!=1 and d<=degF(rem):
        cands=[0b11] if d==1 else range((1<<d)|1,1<<(d+1),2)
        for q in cands:
            if degF(q)!=d or not is_irred_rabin(q): continue
            while rem!=1 and modF(rem,q)==0:
                facs[q]=facs.get(q,0)+1
                rem=divmodF(rem,q)[0]
        d+=1
    if rem!=1:
        assert is_irred_rabin(rem)
        facs[rem]=facs.get(rem,0)+1
    return facs
def prodF(facs):
    r=1
    for q,m in facs.items():
        for _ in range(m): r=mulF(r,q)
    return r
def to_strF(p):
    if p==0: return "0"
    out=[]
    for i in range(degF(p)+1):
        if (p>>i)&1: out.append("1" if i==0 else ("Y" if i==1 else "Y^%d"%i))
    return " + ".join(out)

# ---------------- Z[X] ----------------
def v2_int(n):
    assert n!=0
    v,m=0,abs(n)
    while m%2==0: m//=2; v+=1
    return v
def v0_poly(a):
    nz=[x for x in a if x!=0]
    assert nz, "v0(0) indefinie"
    return min(v2_int(x) for x in nz)
def poly_deg(c):
    d=len(c)-1
    while d>0 and c[d]==0: d-=1
    return d
def poly_add(a,b):
    n=max(len(a),len(b))
    return [(a[i] if i<len(a) else 0)+(b[i] if i<len(b) else 0) for i in range(n)]
def poly_mul(a,b):
    r=[0]*(len(a)+len(b)-1)
    for i,ca in enumerate(a):
        if ca:
            for j,cb in enumerate(b):
                if cb: r[i+j]+=ca*cb
    while len(r)>1 and r[-1]==0: r.pop()
    return r
def poly_divmod(a,b):
    assert any(b)
    db=poly_deg(b); assert b[db]==1
    r=list(a)
    while len(r)<len(b): r.append(0)
    q=[0]*(max(0,len(a)-db)+1)
    for k in range(len(a)-1,db-1,-1):
        if k<len(r) and r[k]!=0:
            cc=r[k]; q[k-db]=cc
            for j in range(db+1): r[k-db+j]-=cc*b[j]
    while len(r)>1 and r[-1]==0: r.pop()
    while len(q)>1 and q[-1]==0: q.pop()
    return q,r
def expansion_phi_adique(T,phi):
    assert phi and phi[-1]==1
    coeffs,reste=[],list(T)
    while True:
        q,r=poly_divmod(reste,phi); coeffs.append(r)
        if len(q)==1 and q[0]==0: break
        reste=q; assert len(coeffs)<=100
    return coeffs
def verifier_expansion(T,phi,coeffs):
    recon,puis=[0],[1]
    for a in coeffs:
        recon=poly_add(recon,poly_mul(a,puis)); puis=poly_mul(puis,phi)
    while len(recon)<len(T): recon.append(0)
    while len(recon)>1 and recon[-1]==0: recon.pop()
    t=list(T)
    while len(t)>1 and t[-1]==0: t.pop()
    assert recon==t, "Expansion phi-adique ne reconstruit pas T"
    return True
def polygone_newton(points):
    # points : liste de (x, y) avec x int, y Fraction. Enveloppe inferieure.
    hull=[]
    for p in points:
        while len(hull)>=2:
            x1,y1=hull[-2]; x2,y2=hull[-1]; x3,y3=p
            if Fraction(y2-y1,x2-x1)>=Fraction(y3-y2,x3-x2): hull.pop()
            else: break
        hull.append(p)
    return hull
def v1_poly(a,phi1,lambda1):
    assert any(a)
    v1_phi=v0_poly(phi1)+lambda1
    coeffs=expansion_phi_adique(a,phi1)
    vals=[v0_poly(cj)+j*v1_phi for j,cj in enumerate(coeffs) if any(cj)]
    assert vals
    return min(vals)
def v1_decomposition(a,phi1,lambda1):
    v1_phi=v0_poly(phi1)+lambda1
    coeffs=expansion_phi_adique(a,phi1)
    det=[]
    for t,ct in enumerate(coeffs):
        if any(ct): det.append((t,ct,v0_poly(ct),v0_poly(ct)+t*v1_phi))
    v1=min(d[3] for d in det); tmin=[d[0] for d in det if d[3]==v1]
    return v1,det,tmin
def validate_exponents(tmin,t0,e1):
    for t in tmin:
        if (t-t0)%e1!=0:
            raise AssertionError("Fausse normalisation (t-t0)%%e1")
    return True
def red1_poly(a,phi1,lambda1):
    assert any(a)
    d=poly_deg(phi1); lam=Fraction(lambda1); e1=lam.denominator
    v1,det,tmin=v1_decomposition(a,phi1,lambda1); t0=min(tmin)
    validate_exponents(tmin,t0,e1)
    phi_mod2=0
    for i,cc in enumerate(phi1):
        if cc%2==1: phi_mod2^=(1<<i)
    if d==1:
        P=0
        for (t,ct,v0ct,_) in det:
            if t in tmin:
                assert (ct[0]//(2**v0ct))%2==1
                P^=(1<<((t-t0)//e1))
        return P,t0,e1,v1
    else:
        Pd={}
        for (t,ct,v0ct,_) in det:
            if t in tmin:
                ad=[(cc//(2**v0ct)) if cc!=0 else 0 for cc in ct]
                am=0
                for j,cc in enumerate(ad):
                    if cc%2==1: am^=(1<<j)
                coeff=modF(am,phi_mod2) if phi_mod2!=0 else am
                e=(t-t0)//e1; Pd[e]=Pd.get(e,0)^coeff
        return Pd,t0,e1,v1
def reduce_P_mod_psi1(P,psi1,phi1):
    d=poly_deg(phi1)
    if d==1:
        if P==0: return 0
        _,r=divmodF(P,psi1); return r

            s=0
            for coeff in P.values(): s^=coeff
            return s
        raise NotImplementedError("Reduction mod psi1 deg>1 pour deg phi1>1 : ouvert")

def t1_of_j_audit(s1, ell1, u_j, e1):
    """t1(j) = (s1 - ell1*u_j)/e1. Si s1/ell1/u_j manquants ou t1 non entier, NON DETERMINE.
       Le point du polygone est conserve, seul le coefficient est NON DETERMINE."""
    if s1 is None or ell1 is None or u_j is None:
        return ("NON DETERMINE", None)
    numF=Fraction(s1) - Fraction(ell1)*Fraction(u_j)
    if numF.denominator!=1: return ("NON DETERMINE", None)
    if (numF.numerator % e1)!=0: return ("NON DETERMINE", None)
    t1=numF.numerator//e1
    if t1<0: return ("NON DETERMINE", None)
    return ("OK", t1)

                    print("[V5.4.4] Residuel inseparable : etape d'ordre superieur necessaire (non ECHEC automatique).")
            segs.append((x0,x1,pente,R2,None,statut))
    return ("OK", points, hull, segs)

def residuel_ordre1(phi,T):
    d=poly_deg(phi); coeffs=expansion_phi_adique(T,phi)
    assert verifier_expansion(T,phi,coeffs)
    pts=[(i,v0_poly(a)) for i,a in enumerate(coeffs) if any(a)]
    hull=polygone_newton(pts); segs=[]
    for k in range(len(hull)-1):
        x0,y0=hull[k]; x1,y1=hull[k+1]; L=x1-x0; Hh=y0-y1
        dd=math.gcd(L,Hh) if Hh!=0 else L; e=L//dd if dd else L
        pente=Fraction(-Hh,L)
        if d==1:
            res=0
            for i in range(x0,x1+1):
                y_exp=Fraction(y0)-Fraction(Hh*(i-x0),L)
                if y_exp.denominator==1 and (i,int(y_exp)) in pts and (i-x0)%e==0:
                    res^=(1<<((i-x0)//e))
            segs.append((x0,x1,pente,res))
        else:
            phi_mod2=0
            for i,cc in enumerate(phi):
                if cc%2==1: phi_mod2^=(1<<i)
            rl=[]
            for i in range(x0,x1+1):
                y_exp=Fraction(y0)-Fraction(Hh*(i-x0),L)
                if y_exp.denominator==1 and (i,int(y_exp)) in pts and (i-x0)%e==0:
                    a=coeffs[i]; vv=v0_poly(a)
                    ad=[(cc//(2**vv)) if cc!=0 else 0 for cc in a]
                    am=0
                    for j,cc in enumerate(ad):
                        if cc%2==1: am^=(1<<j)
                    rl.append(((i-x0)//e, modF(am,phi_mod2) if phi_mod2!=0 else am))
            segs.append((x0,x1,pente,rl))
    return segs

    if len(hull)!=2: return False,det
    (x0,y0),(x1,y1)=hull[0],hull[1]
    pente=Fraction(-(y0-y1),x1-x0); det["pente_phi2"]=pente
    if pente!=Fraction(-Fraction(lambda1)): return False,det
    segs=residuel_ordre1(phi1,phi2)
    det["segments_phi2"]=segs
    if len(segs)!=1: return False,det
    _,_,_,res_phi2=segs[0]
    det["residuel_phi2"]=res_phi2; det["psi1_attendu"]=psi1
    if not (isinstance(res_phi2,int) and isinstance(psi1,int)): return False,det
    if res_phi2!=psi1: return False,det
    if ((psi1>>degF(psi1))&1)!=1: return False,det
    lam=Fraction(lambda1); h1=lam.numerator; e1=lam.denominator
    if e1!=1: return False,det
    diff=[phi2[i]-(phi1[i] if i<len(phi1) else 0) for i in range(max(len(phi2),len(phi1)))]
    while len(diff)>1 and diff[-1]==0: diff.pop()
    det["diff_phi2_phi1"]=diff
    if poly_deg(diff)!=0: return False,det
    dval=diff[0]
    if v2_int(dval)!=h1: return False,det
    if abs(dval)//(2**h1)%2!=1: return False,det
    deg_attendu=poly_deg(phi1)*e1*degF(psi1)
    det["deg_attendu"]=deg_attendu
    if det["degre"]!=deg_attendu: return False,det
    return True,det

def taylor_voie_B(T,phi):
    assert poly_deg(phi)==1 and phi[1]==1
    a0=-phi[0]
    b=[]
    for j in range(len(T)):
        s=0
        for i in range(j,len(T)):
            s+=T[i]*math.comb(i,j)*(a0**(i-j))
        b.append(s)
    while len(b)>1 and b[-1]==0: b.pop()
    return b

def charger_T():
    if os.path.isfile(FIC_T):
        coeffs={}
        with open(FIC_T,"r",encoding="utf-8") as f:
            for line in f:
                line=line.strip()

        assert sorted(coeffs.keys())==list(range(40))
        c=[coeffs[i] for i in range(40)]; src=FIC_T
    else:
        H=None

            if os.path.isfile(cand):
                with open(cand,"r",encoding="utf-8") as f: texte=f.read()
                for line in texte.splitlines():

                else: continue
                s=hline.replace(" - "," +- ").replace(" + "," + ")
                parts=[p.strip() for p in s.split("+") if p.strip()!=""]
                assert len(parts)==40; H={}
                for p in parts:
                    p=p.strip(); neg=p.startswith("-")
                    if neg: p=p[1:].strip()
                    m=re.match(r"(\d+)\*r\*\*(\d+)$",p)
                    if m: coeff,dg=int(m.group(1)),int(m.group(2))
                    else:
                        m=re.match(r"(\d+)\*r$",p)
                        if m: coeff,dg=int(m.group(1)),1

                    if neg: coeff=-coeff

                src=cand; break

        c=[H[i]*(lc**(38-i)) for i in range(39)]+[1]

    ser="\n".join(str(c[i]) for i in range(39,-1,-1))+"\n"
    assert hashlib.sha256(ser.encode("utf-8")).hexdigest()==SHA_T_CANON
    return c,src

def verifier_branches_completes(branches_observes, couples_attendus):
    """Calcule branches_completes ou retourne NON DETERMINE.
       branches_observes : dict q -> liste de couples (pente, residuel) observes.
       couples_attendus : liste de couples (pente, residuel) attendus d'apres les rapports archives,
       ou None si non fournis. Si couples_attendus is None, retourne NON DETERMINE (non False en dur)."""
    if couples_attendus is None:
        return ("NON DETERMINE", None, "Couples attendus archives non fournis.")
    # Verifier que chaque couple attendu est present dans au moins une branche observee.
    for (p_att, r_att) in couples_attendus:
        trouve=False
        for q in branches_observes:
            if (p_att, r_att) in branches_observes[q]:
                trouve=True; break
        if not trouve:
            return ("ECHEC", False, "Couple attendu (%s,%s) absent."%(p_att,r_att))
    return ("OK", True, "Tous les couples attendus presents.")
