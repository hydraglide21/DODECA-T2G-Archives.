class Resultat:
    def __init__(self, nom, statut, detail="", obligatoire=True):
        assert statut in (OK, ECHEC, NON_DET)
        self.nom = nom; self.statut = statut; self.detail = detail; self.obligatoire = obligatoire
    def __repr__(self):
        return "%s : %s (%s)" % (self.nom, self.statut, self.detail)

def statut_global(resultats):
    """ECHEC si un obligatoire refute ; PASS PARTIEL si tests disponibles OK mais indetermines ;
       PASS uniquement si toutes validations obligatoires etablies."""
    # Un NON DETERMINE n'est jamais ajoute aux echecs.
    echecs = [r for r in resultats if r.statut == ECHEC]
    indet = [r for r in resultats if r.statut == NON_DET]
    if any(r.statut == ECHEC and r.obligatoire for r in resultats):
        return ECHEC
    if echecs:
        # Echec non obligatoire : on le signale comme ECHEC global si un test execute est refute.
        # Regle V5.4.5 : ECHEC si au moins un test obligatoire est refute ; ici on durcit :
        # tout ECHEC execute entraine ECHEC global, car un test refute ne peut pas etre ignore.
        return ECHEC
    if indet:
        return "PASS PARTIEL"
    return "PASS"

# ---------------- F2[X] ----------------
def degF(p): return p.bit_length()-1 if p else -1
def mulF(a,b):
    r=0
    while b:
        if b&1: r^=a
        a<<=1; b>>=1
    return r
def divmodF(a,b):
    assert b!=0
    q,r,db=0,a,degF(b)
    while r and degF(r)>=db:
        s=degF(r)-db; q^=(1<<s); r^=(b<<s)
    return q,r
def modF(a,b): return divmodF(a,b)[1]
def gcdF(a,b):
    while b: a,b=b,modF(a,b)
    return a
def derivF(p):
    r=0
    for i in range(degF(p)+1):
        if ((p>>i)&1) and (i%2==1): r^=(1<<(i-1))
    return r
def pow_x_2n_mod(p,n):
    r=0b10
    for _ in range(n): r=modF(mulF(r,r),p)
    return r
def prime_divisors(n):
    res,d,m=[],2,n
    while d*d<=m:
        if m%d==0:
            res.append(d)
            while m%d==0: m//=d
        d+=1 if d==2 else 2
    if m>1: res.append(m)
    return res
def is_irred_rabin(p):
    n=degF(p)
    if n<=0: return False
    if ((p>>n)&1)!=1: return False
    if n==1: return True
    if (p&1)==0: return False
    exp=modF(0b10,p)
    if pow_x_2n_mod(p,n)!=exp: return False
    for ell in prime_divisors(n):
        e=n//ell
        if gcdF(pow_x_2n_mod(p,e)^exp,p)!=1: return False
    return True
def factor_F2(p):
    assert p!=0
    if p==1: return {}
    facs,rem={},p
    while rem!=1 and (rem&1)==0:
        assert modF(rem,0b10)==0
        facs[0b10]=facs.get(0b10,0)+1
        rem=divmodF(rem,0b10)[0]
    d=1
    while rem!=1 and d<=degF(rem):
        cands=[0b11] if d==1 else range((1<<d)|1,1<<(d+1),2)
        for q in cands:
            if degF(q)!=d or not is_irred_rabin(q): continue
            while rem!=1 and modF(rem,q)==0:
                facs[q]=facs.get(q,0)+1
                rem=divmodF(rem,q)[0]
        d+=1
    if rem!=1:
        assert is_irred_rabin(rem)
        facs[rem]=facs.get(rem,0)+1
    return facs
def prodF(facs):
    r=1
    for q,m in facs.items():
        for _ in range(m): r=mulF(r,q)
    return r
def to_strF(p):
    if p==0: return "0"
    out=[]
    for i in range(degF(p)+1):
        if (p>>i)&1: out.append("1" if i==0 else ("Y" if i==1 else "Y^%d"%i))
    return " + ".join(out)

# ---------------- Z[X] ----------------
def v2_int(n):
    assert n!=0
    v,m=0,abs(n)
    while m%2==0: m//=2; v+=1
    return v
def v0_poly(a):
    nz=[x for x in a if x!=0]
    assert nz, "v0(0) indefinie"
    return min(v2_int(x) for x in nz)
def poly_deg(c):
    d=len(c)-1
    while d>0 and c[d]==0: d-=1
    return d
def poly_add(a,b):
    n=max(len(a),len(b))
    return [(a[i] if i<len(a) else 0)+(b[i] if i<len(b) else 0) for i in range(n)]
def poly_mul(a,b):
    r=[0]*(len(a)+len(b)-1)
    for i,ca in enumerate(a):
        if ca:
            for j,cb in enumerate(b):
                if cb: r[i+j]+=ca*cb
    while len(r)>1 and r[-1]==0: r.pop()
    return r
def poly_divmod(a,b):
    assert any(b)
    db=poly_deg(b); assert b[db]==1
    r=list(a)
    while len(r)<len(b): r.append(0)
    q=[0]*(max(0,len(a)-db)+1)
    for k in range(len(a)-1,db-1,-1):
        if k<len(r) and r[k]!=0:
            cc=r[k]; q[k-db]=cc
            for j in range(db+1): r[k-db+j]-=cc*b[j]
    while len(r)>1 and r[-1]==0: r.pop()
    while len(q)>1 and q[-1]==0: q.pop()
    return q,r
def expansion_phi_adique(T,phi):
    assert phi and phi[-1]==1
    coeffs,reste=[],list(T)
    while True:
        q,r=poly_divmod(reste,phi); coeffs.append(r)
        if len(q)==1 and q[0]==0: break
        reste=q; assert len(coeffs)<=100
    return coeffs
def verifier_expansion(T,phi,coeffs):
    recon,puis=[0],[1]
    for a in coeffs:
        recon=poly_add(recon,poly_mul(a,puis)); puis=poly_mul(puis,phi)
    while len(recon)<len(T): recon.append(0)
    while len(recon)>1 and recon[-1]==0: recon.pop()
    t=list(T)
    while len(t)>1 and t[-1]==0: t.pop()
    assert recon==t, "Expansion phi-adique ne reconstruit pas T"
    return True
def polygone_newton(points):
    hull=[]
    for p in points:
        while len(hull)>=2:
            x1,y1=hull[-2]; x2,y2=hull[-1]; x3,y3=p
            if Fraction(y2-y1,x2-x1)>=Fraction(y3-y2,x3-x2): hull.pop()
            else: break
        hull.append(p)
    return hull
def v1_poly(a,phi1,lambda1):
    assert any(a)
    v1_phi=v0_poly(phi1)+lambda1
    coeffs=expansion_phi_adique(a,phi1)
    vals=[v0_poly(cj)+j*v1_phi for j,cj in enumerate(coeffs) if any(cj)]
    assert vals
    return min(vals)
def v1_decomposition(a,phi1,lambda1):
    v1_phi=v0_poly(phi1)+lambda1
    coeffs=expansion_phi_adique(a,phi1)
    det=[]
    for t,ct in enumerate(coeffs):
        if any(ct): det.append((t,ct,v0_poly(ct),v0_poly(ct)+t*v1_phi))
    v1=min(d[3] for d in det); tmin=[d[0] for d in det if d[3]==v1]
    return v1,det,tmin

# ---------- Validateur central anti-flottant ----------
def valider_pas_de_flottant_polygone(points, hull=None, pentes=None):
    """Refuse toute valeur float dans les donnees d'un polygone.
       Retourne (statut, detail). OK si aucune float ; ECHEC si une float trouvee ;
       NON DETERMINE si donnees manquantes."""
    if points is None:
        return (NON_DET, "Points manquants")
    for (x,y) in points:
        if isinstance(x,float) or isinstance(y,float):
            return (ECHEC, "Float trouve dans un point (%r,%r)"%(x,y))
        if not (isinstance(x,int) and isinstance(y,Fraction)):
            # x doit etre int, y Fraction (ou int accepte comme Fraction via conversion).
            if isinstance(y,int):
                pass
            else:
                return (ECHEC, "Type inattendu dans un point (%r,%r)"%(x,y))
    if hull is not None:
        for (x,y) in hull:
            if isinstance(x,float) or isinstance(y,float):
                return (ECHEC, "Float trouve dans un sommet")
    if pentes is not None:
        for p in pentes:
            if isinstance(p,float):
                return (ECHEC, "Pente float trouvee : %r"%p)
            if not isinstance(p,Fraction):
                return (ECHEC, "Pente non Fraction : %r"%p)
    return (OK, "Aucun flottant, types exacts")

def valider_normalisation_t1(t1_propose, s1, ell1, u_j, e1):
    """Validateur de normalisation : rejette la substitution t1 -> t0.
       Calcule t1 attendu par la formule GMN si les donnees sont disponibles,
       sinon NON DETERMINE. Compare au t1 propose."""
    if s1 is None or ell1 is None or u_j is None or e1 is None:
        return (NON_DET, "Donnees t1 manquantes")
    numF = Fraction(s1) - Fraction(ell1)*Fraction(u_j)
    if numF.denominator != 1 or (numF.numerator % e1) != 0 or (numF.numerator//e1) < 0:
        return (NON_DET, "t1 non entier / negatif")
    t1_attendu = numF.numerator//e1
    if t1_propose != t1_attendu:
        return (ECHEC, "t1 propose %s != t1 attendu %s"%(t1_propose, t1_attendu))
    return (OK, "t1 = %s"%t1_attendu)

# ---------- 5. Authentification T : fonction unique ----------
def valider_authenticite_T(coefficients, sha_attendu):
    """Verifie degre 39, monicite, serialisation canonique, SHA-256, comparaison.
       Retourne (statut, detail). OK ou ECHEC. Ne modifie jamais le T authentique."""
    try:
        if coefficients is None or len(coefficients)!=40:
            return (ECHEC, "Longueur != 40")
        # Degre 39 : dernier coefficient non nul, tous non nuls (pour notre T).
        if poly_deg(coefficients)!=39:
            return (ECHEC, "Degre != 39")
        if coefficients[39]!=1:
            return (ECHEC, "Non monique (c39 != 1)")
        ser="\n".join(str(coefficients[i]) for i in range(39,-1,-1))+"\n"
        h=hashlib.sha256(ser.encode("utf-8")).hexdigest()
        if h!=sha_attendu:
            return (ECHEC, "SHA != attendu (calcule %s)"%h)
        return (OK, "Degre 39, monique, SHA OK")
    except Exception as e:
        return (ECHEC, "Exception : %s"%e)

# ---------- 6. Certification phi2 : CAS PARTICULIER e1==1 ----------
def certifier_phi2_cas_particulier_e1_1(phi1,lambda1,psi1,phi2):
    """CERTIFICATION PHI2 : CAS PARTICULIER e1=1. Ne pas la presenter comme procedure GMN generale.
       Si e1 != 1, retourne NON DETERMINE, sauf si une hypothese precise est refutee (alors ECHEC)."""
    lam=Fraction(lambda1)
    if lam.denominator!=1:
        return (NON_DET, {"raison":"e1 != 1 : certification generale non implementee"})
    # Cas e1==1 : procedure du premier ordre (verifiable).
    det={}
    det["degre"]=poly_deg(phi2); det["monique"]=(phi2[-1]==1)
    if not det["monique"] or det["degre"]<1:
        return (ECHEC, det)
    exp=expansion_phi_adique(phi2,phi1)
    det["expansion_phi1"]=exp
    if not verifier_expansion(phi2,phi1,exp):
        return (ECHEC, det)
    pts=[(i,v0_poly(a)) for i,a in enumerate(exp) if any(a)]
    hull=polygone_newton(pts)
    det["points_phi2"]=pts; det["hull_phi2"]=hull
    if len(hull)!=2:
        return (ECHEC, det)
    (x0,y0),(x1,y1)=hull[0],hull[1]
    pente=Fraction(-(y0-y1),x1-x0); det["pente_phi2"]=pente
    if pente!=Fraction(-Fraction(lambda1)):
        return (ECHEC, det)
    segs=residuel_ordre1(phi1,phi2)
    det["segments_phi2"]=segs
    if len(segs)!=1:
        return (ECHEC, det)
    _,_,_,res_phi2=segs[0]
    det["residuel_phi2"]=res_phi2; det["psi1_attendu"]=psi1
    if not (isinstance(res_phi2,int) and isinstance(psi1,int)):
        return (NON_DET, det)
    if res_phi2!=psi1:
        return (ECHEC, det)
    if ((psi1>>degF(psi1))&1)!=1:
        return (ECHEC, det)
    h1=lam.numerator; e1=lam.denominator
    diff=[phi2[i]-(phi1[i] if i<len(phi1) else 0) for i in range(max(len(phi2),len(phi1)))]
    while len(diff)>1 and diff[-1]==0: diff.pop()
    det["diff_phi2_phi1"]=diff
    if poly_deg(diff)!=0:
        return (ECHEC, det)
    dval=diff[0]
    if v2_int(dval)!=h1:
        return (ECHEC, det)
    if abs(dval)//(2**h1)%2!=1:
        return (ECHEC, det)
    deg_attendu=poly_deg(phi1)*e1*degF(psi1)
    det["deg_attendu"]=deg_attendu
    if det["degre"]!=deg_attendu:
        return (ECHEC, det)
    return (OK, det)

def residuel_ordre1(phi,T):
    d=poly_deg(phi); coeffs=expansion_phi_adique(T,phi)
    assert verifier_expansion(T,phi,coeffs)
    pts=[(i,v0_poly(a)) for i,a in enumerate(coeffs) if any(a)]
    hull=polygone_newton(pts); segs=[]
    for k in range(len(hull)-1):
        x0,y0=hull[k]; x1,y1=hull[k+1]; L=x1-x0; Hh=y0-y1
        dd=math.gcd(L,Hh) if Hh!=0 else L; e=L//dd if dd else L
        pente=Fraction(-Hh,L)
        if d==1:
            res=0
            for i in range(x0,x1+1):
                y_exp=Fraction(y0)-Fraction(Hh*(i-x0),L)
                if y_exp.denominator==1 and (i,int(y_exp)) in pts and (i-x0)%e==0:
                    res^=(1<<((i-x0)//e))
            segs.append((x0,x1,pente,res))
        else:
            phi_mod2=0
            for i,cc in enumerate(phi):
                if cc%2==1: phi_mod2^=(1<<i)
            rl=[]
            for i in range(x0,x1+1):
                y_exp=Fraction(y0)-Fraction(Hh*(i-x0),L)
                if y_exp.denominator==1 and (i,int(y_exp)) in pts and (i-x0)%e==0:
                    a=coeffs[i]; vv=v0_poly(a)
                    ad=[(cc//(2**vv)) if cc!=0 else 0 for cc in a]
                    am=0
                    for j,cc in enumerate(ad):
                        if cc%2==1: am^=(1<<j)
                    rl.append(((i-x0)//e, modF(am,phi_mod2) if phi_mod2!=0 else am))
            segs.append((x0,x1,pente,rl))
    return segs

# ---------- 7. Audit GMN ordre 2 : prudence, aucune imposition ----------
def audit_invariants_type(phi1,lambda1,psi1):
    lam=Fraction(lambda1); e1=lam.denominator; h1=lam.numerator; f1=degF(psi1)
    params={"e1":e1,"h1":h1,"f1":f1}
    if math.gcd(h1,e1)!=1:
        return (ECHEC, params, ["gcd(h1,e1) != 1"])

def t1_of_j_audit(s1, ell1, u_j, e1):
    if s1 is None or ell1 is None or u_j is None:
        return (NON_DET, None)
    numF=Fraction(s1)-Fraction(ell1)*Fraction(u_j)

def taylor_voie_B(T,phi):
    assert poly_deg(phi)==1 and phi[1]==1
    a0=-phi[0]
    b=[]
    for j in range(len(T)):
        s=0
        for i in range(j,len(T)):
            s+=T[i]*math.comb(i,j)*(a0**(i-j))
        b.append(s)
    while len(b)>1 and b[-1]==0: b.pop()
    return b
