def check_no_bool_no_float(x):

class Resultat:

        assert statut in (OK, ECHEC, NON_DET, "PASS PARTIEL", "PASS")
        self.nom = nom; self.statut = statut; self.detail = detail; self.obligatoire = obligatoire

def statut_global(resultats):
    """ECHEC si obligatoire refute ; PASS PARTIEL si controles disponibles passent mais
       obligatoires indetermines, ou si un non-obligatoire echoue (signale, non confondu) ;
       PASS uniquement si toutes validations obligatoires etablies et reussissent.
       Ne jamais convertir NON DETERMINE en ECHEC."""
    mand_echec = [r for r in resultats if r.obligatoire and r.statut == ECHEC]
    if mand_echec:
        return ECHEC
    mand_nd = [r for r in resultats if r.obligatoire and r.statut == NON_DET]
    nonmand_echec = [r for r in resultats if (not r.obligatoire) and r.statut == ECHEC]
    nonmand_nd = [r for r in resultats if (not r.obligatoire) and r.statut == NON_DET]
    if mand_nd or nonmand_echec or nonmand_nd:
        return "PASS PARTIEL"
    return "PASS"

# ---------------- F2[X] ----------------
def degF(p): return p.bit_length()-1 if p else -1
def mulF(a,b):
    r=0
    while b:
        if b&1: r^=a
        a<<=1; b>>=1
    return r
def divmodF(a,b):
    assert b!=0
    q,r,db=0,a,degF(b)
    while r and degF(r)>=db:
        s=degF(r)-db; q^=(1<<s); r^=(b<<s)
    return q,r
def modF(a,b): return divmodF(a,b)[1]
def gcdF(a,b):
    while b: a,b=b,modF(a,b)
    return a
def derivF(p):
    r=0
    for i in range(degF(p)+1):
        if ((p>>i)&1) and (i%2==1): r^=(1<<(i-1))
    return r
def pow_x_2n_mod(p,n):
    r=0b10
    for _ in range(n): r=modF(mulF(r,r),p)
    return r
def prime_divisors(n):
    res,d,m=[],2,n
    while d*d<=m:
        if m%d==0:
            res.append(d)
            while m%d==0: m//=d
        d+=1 if d==2 else 2
    if m>1: res.append(m)
    return res
def is_irred_rabin(p):
    n=degF(p)
    if n<=0: return False
    if ((p>>n)&1)!=1: return False
    if n==1: return True
    if (p&1)==0: return False
    exp=modF(0b10,p)
    if pow_x_2n_mod(p,n)!=exp: return False
    for ell in prime_divisors(n):
        e=n//ell
        if gcdF(pow_x_2n_mod(p,e)^exp,p)!=1: return False
    return True
def factor_F2(p):
    assert p!=0
    if p==1: return {}
    facs,rem={},p
    while rem!=1 and (rem&1)==0:
        assert modF(rem,0b10)==0
        facs[0b10]=facs.get(0b10,0)+1
        rem=divmodF(rem,0b10)[0]
    d=1
    while rem!=1 and d<=degF(rem):
        cands=[0b11] if d==1 else range((1<<d)|1,1<<(d+1),2)
        for q in cands:
            if degF(q)!=d or not is_irred_rabin(q): continue
            while rem!=1 and modF(rem,q)==0:
                facs[q]=facs.get(q,0)+1
                rem=divmodF(rem,q)[0]
        d+=1
    if rem!=1:
        assert is_irred_rabin(rem)
        facs[rem]=facs.get(rem,0)+1
    return facs
def prodF(facs):
    r=1
    for q,m in facs.items():
        for _ in range(m): r=mulF(r,q)
    return r
def to_strF(p):
    if p==0: return "0"
    out=[]
    for i in range(degF(p)+1):
        if (p>>i)&1: out.append("1" if i==0 else ("Y" if i==1 else "Y^%d"%i))
    return " + ".join(out)

# ---------------- Z[X] ----------------
def v2_int(n):
    # n doit etre int strict, non bool, non nul.
    if type(n) is bool or not type(n) is int:
        raise TypeError("v2_int exige int strict, non bool/float")
    assert n!=0
    v,m=0,abs(n)
    while m%2==0: m//=2; v+=1
    return v
def v0_poly(a):
    nz=[x for x in a if x!=0]
    assert nz, "v0(0) indefinie"
    for x in nz:
        if type(x) is bool or not type(x) is int:
            raise TypeError("v0_poly exige coefficients int stricts")
    return min(v2_int(x) for x in nz)
def poly_deg(c):
    d=len(c)-1
    while d>0 and c[d]==0: d-=1
    return d
def poly_add(a,b):
    n=max(len(a),len(b))
    return [(a[i] if i<len(a) else 0)+(b[i] if i<len(b) else 0) for i in range(n)]
def poly_mul(a,b):
    r=[0]*(len(a)+len(b)-1)
    for i,ca in enumerate(a):
        if ca:
            for j,cb in enumerate(b):
                if cb: r[i+j]+=ca*cb
    while len(r)>1 and r[-1]==0: r.pop()
    return r
def poly_divmod(a,b):
    assert any(b)
    db=poly_deg(b); assert b[db]==1
    r=list(a)
    while len(r)<len(b): r.append(0)
    q=[0]*(max(0,len(a)-db)+1)
    for k in range(len(a)-1,db-1,-1):
        if k<len(r) and r[k]!=0:
            cc=r[k]; q[k-db]=cc
            for j in range(db+1): r[k-db+j]-=cc*b[j]
    while len(r)>1 and r[-1]==0: r.pop()
    while len(q)>1 and q[-1]==0: q.pop()
    return q,r
def expansion_phi_adique(T,phi):
    assert phi and phi[-1]==1
    coeffs,reste=[],list(T)
    while True:
        q,r=poly_divmod(reste,phi); coeffs.append(r)
        if len(q)==1 and q[0]==0: break
        reste=q; assert len(coeffs)<=100
    return coeffs
def verifier_expansion(T,phi,coeffs):
    recon,puis=[0],[1]
    for a in coeffs:
        recon=poly_add(recon,poly_mul(a,puis)); puis=poly_mul(puis,phi)
    while len(recon)<len(T): recon.append(0)
    while len(recon)>1 and recon[-1]==0: recon.pop()
    t=list(T)
    while len(t)>1 and t[-1]==0: t.pop()
    assert recon==t, "Expansion phi-adique ne reconstruit pas T"
    return True
def polygone_newton(points):
    # points : (x int strict, y Fraction). Controle des types.
    for (x,y) in points:
        if type(x) is bool or not type(x) is int:
            raise TypeError("Abscisse doit etre int strict")
        if type(y) is bool or isinstance(y,float) or not (type(y) is int or type(y) is Fraction):
            raise TypeError("Ordonnee doit etre int/Fraction, non float/bool")
    hull=[]
    for p in points:
        while len(hull)>=2:
            x1,y1=hull[-2]; x2,y2=hull[-1]; x3,y3=p
            if Fraction(y2-y1,x2-x1)>=Fraction(y3-y2,x3-x2): hull.pop()
            else: break
        hull.append(p)
    return hull
def v1_poly(a,phi1,lambda1):
    assert any(a)
    v1_phi=v0_poly(phi1)+lambda1
    coeffs=expansion_phi_adique(a,phi1)
    vals=[v0_poly(cj)+j*v1_phi for j,cj in enumerate(coeffs) if any(cj)]
    assert vals
    return min(vals)
def v1_decomposition(a,phi1,lambda1):
    v1_phi=v0_poly(phi1)+lambda1
    coeffs=expansion_phi_adique(a,phi1)
    det=[]
    for t,ct in enumerate(coeffs):
        if any(ct): det.append((t,ct,v0_poly(ct),v0_poly(ct)+t*v1_phi))
    v1=min(d[3] for d in det); tmin=[d[0] for d in det if d[3]==v1]
    return v1,det,tmin

# ---------- Validateur central anti-flottant / anti-bool : points, sommets, exposants, valuations ----------
def valider_types_polygone(points, hull=None, pentes=None, exposants=None, valuations=None):
    """Refuse float et bool dans points, sommets, pentes, exposants, valuations.
       Retourne (statut, detail)."""
    if points is None:
        return (NON_DET, "Points manquants")
    for (x,y) in points:
        if isinstance(x,float) or type(x) is bool:
            return (ECHEC, "Abscisse float/bool : %r"%x)
        if isinstance(y,float) or type(y) is bool:
            return (ECHEC, "Ordonnee float/bool : %r"%y)
        if not (type(x) is int and (type(y) is int or type(y) is Fraction)):
            return (ECHEC, "Type inattendu point (%r,%r)"%(x,y))
    if hull is not None:
        for (x,y) in hull:
            if isinstance(x,float) or type(x) is bool or isinstance(y,float) or type(y) is bool:
                return (ECHEC, "Sommet float/bool")
    if pentes is not None:
        for p in pentes:
            if isinstance(p,float) or type(p) is bool:
                return (ECHEC, "Pente float/bool : %r"%p)
            if not type(p) is Fraction:
                return (ECHEC, "Pente non Fraction : %r"%p)
    if exposants is not None:
        for e in exposants:
            if isinstance(e,float) or type(e) is bool or not type(e) is int:
                return (ECHEC, "Exposant float/bool/non-int : %r"%e)
    if valuations is not None:
        for v in valuations:
            if isinstance(v,float) or type(v) is bool or not (type(v) is int or type(v) is Fraction):
                return (ECHEC, "Valuation float/bool : %r"%v)
    return (OK, "Types exacts, aucun float/bool")

# ---------- 2. Validateur ordonnees GMN : recalcule chaque u_j ----------
def valider_ordonnees_gmn(coeffs_bj, v2_bj_list, v2_phi2, u_j_list):
    """Validateur de production de la formule u_j = v2(b_j) + j*v2(phi2).
       Entrees : coeffs_bj (liste), v2_bj_list (valeurs exactes etablies ou None si manquantes),
       v2_phi2 (valeur exacte etablie ou None), u_j_list (ordonnees calculees proposees).
       - Si v2_phi2 is None ou un v2_bj is None -> NON DETERMINE (donnee manquante).
       - Si un type est float/bool -> ECHEC (type).
       - Sinon recalcule chaque u_j attendue en Fraction et compare exactement.
       Ne jamais assimiler v2(b_j) a la valuation minimale sans justification : v2_bj_list
       doit etre fournie comme valeurs etablies, non derivee ici de v0."""
    if coeffs_bj is None or v2_bj_list is None or u_j_list is None:
        return (NON_DET, "Donnees manquantes (coeffs/v2_bj/u_j)")
    if len(coeffs_bj)!=len(v2_bj_list) or len(coeffs_bj)!=len(u_j_list):
        return (ECHEC, "Longueurs incoherentes")
    if v2_phi2 is None:
        return (NON_DET, "v2(phi2) manquante")
    # Controle types de v2_phi2.
    if isinstance(v2_phi2,float) or type(v2_phi2) is bool:
        return (ECHEC, "v2(phi2) float/bool")
    if not (type(v2_phi2) is int or type(v2_phi2) is Fraction):
        return (ECHEC, "v2(phi2) type inattendu")
    for idx, (bj, v2bj, uj) in enumerate(zip(coeffs_bj, v2_bj_list, u_j_list)):
        if v2bj is None:
            return (NON_DET, "v2(b_%d) manquante"%idx)
        if uj is None:
            return (NON_DET, "u_%d manquante"%idx)
        if isinstance(v2bj,float) or type(v2bj) is bool:
            return (ECHEC, "v2(b_%d) float/bool"%idx)
        if isinstance(uj,float) or type(uj) is bool:
            return (ECHEC, "u_%d float/bool"%idx)
        if not (type(v2bj) is int or type(v2bj) is Fraction):
            return (ECHEC, "v2(b_%d) type inattendu"%idx)
        if not (type(uj) is int or type(uj) is Fraction):
            return (ECHEC, "u_%d type inattendu"%idx)
        # Recalcul exact en Fraction.
        u_attendue = Fraction(v2bj) + Fraction(idx)*Fraction(v2_phi2)
        if Fraction(uj) != u_attendue:
            return (ECHEC, "u_%d : propose %s != attendu %s (v2(b)=%s, v2(phi2)=%s)"%(idx, uj, u_attendue, v2bj, v2_phi2))
    return (OK, "Toutes les ordonnees recalculees et exactes")

# ---------- 3. Validateur normalisation t1 ----------
def valider_t1(s1, ell1, u_j, e1, t1_propose):
    """Validateur de t1(j) = (s1 - ell1*u_j)/e1.
       - e1 doit etre int strict strictement positif. e1==0 -> ECHEC (impossible).
         e1 float/bool -> ECHEC (type). e1 None -> NON DETERMINE.
       - s1/ell1/u_j None -> NON DETERMINE. float/bool -> ECHEC.
       - Calcul en Fraction, integrite controlee, comparaison au t1 propose.
       - t1_propose None -> NON DETERMINE. float/bool -> ECHEC."""
    if e1 is None:
        return (NON_DET, "e1 manquant")
    if isinstance(e1,float) or type(e1) is bool:
        return (ECHEC, "e1 float/bool")
    if not type(e1) is int:
        return (ECHEC, "e1 non int")
    if e1==0:
        return (ECHEC, "e1=0 impossible (division par zero)")
    if e1<0:
        return (ECHEC, "e1 negatif")
    if s1 is None or ell1 is None or u_j is None:
        return (NON_DET, "Donnees t1 manquantes (s1/ell1/u_j)")
    for (nom, val) in [("s1",s1),("ell1",ell1),("u_j",u_j)]:
        if isinstance(val,float) or type(val) is bool:
            return (ECHEC, "%s float/bool"%nom)
        if not (type(val) is int or type(val) is Fraction):
            return (ECHEC, "%s type inattendu"%nom)
    if t1_propose is None:
        return (NON_DET, "t1 propose manquant")
    if isinstance(t1_propose,float) or type(t1_propose) is bool:
        return (ECHEC, "t1 propose float/bool")
    if not type(t1_propose) is int:
        return (ECHEC, "t1 propose non int")
    numF = Fraction(s1) - Fraction(ell1)*Fraction(u_j)
    if numF.denominator!=1:
        return (NON_DET, "t1 non entier (numF=%s)"%numF)
    if (numF.numerator % e1)!=0:
        return (NON_DET, "t1 non entier (division e1)")
    t1_attendu = numF.numerator//e1
    if t1_attendu<0:
        return (NON_DET, "t1 negatif")
    if t1_propose!=t1_attendu:
        return (ECHEC, "t1 propose %s != attendu %s"%(t1_propose, t1_attendu))
    return (OK, "t1 = %s"%t1_attendu)

def validate_exponents(tmin,t0,e1):
    for t in tmin:
        if (t-t0)%e1!=0:
            raise AssertionError("Fausse normalisation (t-t0)%%e1")
    return True
def red1_poly(a,phi1,lambda1):
    assert any(a)
    d=poly_deg(phi1); lam=Fraction(lambda1); e1=lam.denominator
    v1,det,tmin=v1_decomposition(a,phi1,lambda1); t0=min(tmin)
    validate_exponents(tmin,t0,e1)
    phi_mod2=0
    for i,cc in enumerate(phi1):
        if cc%2==1: phi_mod2^=(1<<i)
    if d==1:
        P=0
        for (t,ct,v0ct,_) in det:
            if t in tmin:
                assert (ct[0]//(2**v0ct))%2==1
                P^=(1<<((t-t0)//e1))
        return P,t0,e1,v1
    else:
        Pd={}
        for (t,ct,v0ct,_) in det:
            if t in tmin:
                ad=[(cc//(2**v0ct)) if cc!=0 else 0 for cc in ct]
                am=0
                for j,cc in enumerate(ad):
                    if cc%2==1: am^=(1<<j)
                coeff=modF(am,phi_mod2) if phi_mod2!=0 else am
                e=(t-t0)//e1; Pd[e]=Pd.get(e,0)^coeff
        return Pd,t0,e1,v1
def reduce_P_mod_psi1(P,psi1,phi1):
    d=poly_deg(phi1)
    if d==1:
        if P==0: return 0
        _,r=divmodF(P,psi1); return r
    else:
        if degF(psi1)==1:
            s=0
            for coeff in P.values(): s^=coeff
            return s
        raise NotImplementedError("Reduction mod psi1 deg>1 pour deg phi1>1 : ouvert")

# Etats : ETABLI / CALCULE / NON ETABLI / CONTRADICTION
def audit_invariants_type(phi1,lambda1,psi1):
    """Pour chaque invariant, etat independant. N'impose jamais les formules supposees.
       e1,h1,f1 : CALCULE (derives de lambda1, psi1, verifies).
       ell1, ell'_1, pi1, pi2, Phi1, gamma1, z1, s1, v2 : NON ETABLI sans Def. verifiees."""
    lam=Fraction(lambda1)
    e1=lam.denominator; h1=lam.numerator; f1=degF(psi1)
    invariants={}
    invariants["e1"]={"etat":"CALCULE","valeur":e1,"detail":"denominateur(lambda1)"}
    invariants["h1"]={"etat":"CALCULE","valeur":h1,"detail":"numerateur(lambda1)"}
    invariants["f1"]={"etat":"CALCULE","valeur":f1,"detail":"deg psi1"}
    if math.gcd(h1,e1)!=1:
        invariants["e1"]["etat"]="CONTRADICTION"
        return ("ECHEC", invariants)
    for nom in ["ell1","ell_prime_1","pi1","pi2","Phi1","gamma1","z1","s1","v2"]:
        invariants[nom]={"etat":"NON ETABLI","valeur":None,"detail":"Def. 2.13/2.14/2.20/2.21 non verifiees ; aucune formule supposee"}
    return (NON_DET, invariants)

       Retourne (statut, chemin_trouve, note). Ne remplace jamais silencieusement

    candidats.append(NOM_LOGIQUE_T)
    candidats.append(os.path.join("/mnt/data", NOM_LOGIQUE_T))

    candidats.append(NOM_ENCODE_T)

            else:
                note="Fichier trouve sous nom LOGIQUE (underscores reels)."
            return (OK, cand, note)
    return (NON_DET, None, "Aucun fichier T trouve (ni logique ni encode).")

def valider_authenticite_T(coefficients, sha_attendu):
    """Verifie : types stricts (aucun bool/float), longueur 40, degre 39, monicite,
       serialisation canonique, SHA-256, comparaison. Retourne (statut, detail).
       OK ou ECHEC. Coefficients manquants / mauvaise monicite / bool -> ECHEC."""
    if coefficients is None:
        return (ECHEC, "Coefficients manquants (None)")
    if not isinstance(coefficients, list):
        return (ECHEC, "Coefficients non liste")
    if len(coefficients)!=40:
        return (ECHEC, "Longueur %d != 40 (coefficient manquant ou en trop)"%len(coefficients))
    for idx, c in enumerate(coefficients):
        if isinstance(c,float) or type(c) is bool:
            return (ECHEC, "Coefficient %d float/bool : %r"%(idx,c))
        if not type(c) is int:
            return (ECHEC, "Coefficient %d non int : %r"%(idx,c))
    # Degre 39 : poly_deg doit etre 39 (dernier coeff non nul).
    try:
        if poly_deg(coefficients)!=39:
            return (ECHEC, "Degre %d != 39"%poly_deg(coefficients))
    except Exception as e:
        return (ECHEC, "Degre : %s"%e)
    if coefficients[39]!=1:
        return (ECHEC, "Mauvaise monicite (c39=%s != 1)"%coefficients[39])
    # Serialisation canonique : "\n".join(str(c[i]) for i in 39..0) + "\n".
    # Convention a confronter au fichier source : voir charger_T_authentifie.
    ser="\n".join(str(coefficients[i]) for i in range(39,-1,-1))+"\n"
    h=hashlib.sha256(ser.encode("utf-8")).hexdigest()
    if h!=sha_attendu:
        return (ECHEC, "SHA %s != attendu"%h)
    return (OK, "Degre 39, monique, types stricts, SHA OK")

def charger_T_authentifie():
    """Charge T depuis le fichier source authentique, verifie la convention de serialisation.
       Retourne (statut, coefficients, source, convention). Ne remplace jamais silencieusement
       le fichier authentique par un secours non authentifie : si le fichier T n'existe pas,
       retourne NON DETERMINE (pas de fallback H= silencieux)."""
    st_find, chemin, note = trouver_fichier_T()
    if st_find!=OK:
        return (NON_DET, None, None, note)
    # Lire le fichier et verifier sa structure.
    try:
        with open(chemin,"r",encoding="utf-8") as f:
            texte=f.read()
    except Exception as e:
        return (ECHEC, None, chemin, "Lecture : %s"%e)
    # Deux formes possibles : c_0..c_39 (forme directe) ou H= (forme resultant).
    # La convention SHA canonique est definie sur la serialisation c_39..c_0.
    # Si le fichier est en forme H=, la transformation H -> coefficients doit etre documentee
    # comme DERIVEE, non supposee canonique. Ici, on n'accepte que la forme c_ comme directe.
    # Si le fichier contient des lignes c_, on les parse ; sinon, on retourne NON DETERMINE
    # pour la convention (on ne fabrique pas de coefficients depuis H= sans authentification).
    coeffs_dict={}
    has_c=False
    for line in texte.splitlines():
        line=line.strip()
        if line.startswith("c_"):
            has_c=True
            if "=" in line:
                k,v=line.split("=",1)
                try:
                    coeffs_dict[int(k[2:])]=int(v)
                except Exception:
                    return (ECHEC, None, chemin, "Ligne c_ non entiere")
    if has_c:
        if sorted(coeffs_dict.keys())!=list(range(40)):
            return (ECHEC, None, chemin, "Coefficients c_ incomplets")
        c=[coeffs_dict[i] for i in range(40)]
        # Verifier que la serialisation reconstruite correspond bien au fichier (convention).
        # On reconstruit la serialisation canonique et on verifie le SHA.
        st_auth, det_auth = valider_authenticite_T(c, SHA_T_CANON)
        if st_auth!=OK:
            return (ECHEC, None, chemin, "Authentification : %s"%det_auth)
        return (OK, c, chemin, "Convention c_ directe, serialisation canonique verifiee. %s"%note)
    # Forme H= : on ne la traite pas comme authentique sans verification supplementaire.
    # On retourne NON DETERMINE pour la convention, en signalant la forme.
    if "H=" in texte:
        return (NON_DET, None, chemin, "Fichier en forme H= : convention de serialisation non confrontee directement ; pas de fallback silencieux. %s"%note)
    return (ECHEC, None, chemin, "Forme de fichier T inconnue")

def residuel_ordre1(phi,T):
    d=poly_deg(phi); coeffs=expansion_phi_adique(T,phi)
    assert verifier_expansion(T,phi,coeffs)
    pts=[(i,v0_poly(a)) for i,a in enumerate(coeffs) if any(a)]
    hull=polygone_newton(pts); segs=[]
    for k in range(len(hull)-1):
        x0,y0=hull[k]; x1,y1=hull[k+1]; L=x1-x0; Hh=y0-y1
        dd=math.gcd(L,Hh) if Hh!=0 else L; e=L//dd if dd else L
        pente=Fraction(-Hh,L)
        if d==1:
            res=0
            for i in range(x0,x1+1):
                y_exp=Fraction(y0)-Fraction(Hh*(i-x0),L)
                if y_exp.denominator==1 and (i,int(y_exp)) in pts and (i-x0)%e==0:
                    res^=(1<<((i-x0)//e))
            segs.append((x0,x1,pente,res))
        else:
            phi_mod2=0
            for i,cc in enumerate(phi):
                if cc%2==1: phi_mod2^=(1<<i)
            rl=[]
            for i in range(x0,x1+1):
                y_exp=Fraction(y0)-Fraction(Hh*(i-x0),L)
                if y_exp.denominator==1 and (i,int(y_exp)) in pts and (i-x0)%e==0:
                    a=coeffs[i]; vv=v0_poly(a)
                    ad=[(cc//(2**vv)) if cc!=0 else 0 for cc in a]
                    am=0
                    for j,cc in enumerate(ad):
                        if cc%2==1: am^=(1<<j)
                    rl.append(((i-x0)//e, modF(am,phi_mod2) if phi_mod2!=0 else am))
            segs.append((x0,x1,pente,rl))
    return segs

# ---------- 8. Certification phi2 : CAS PARTICULIER e1==1 ----------
def certifier_phi2_cas_particulier_e1_1(phi1,lambda1,psi1,phi2):
    """CERTIFICATION PHI2 : CAS PARTICULIER e1=1 uniquement.
       Si e1 != 1 -> NON DETERMINE tant que construction generale non demontree."""
    lam=Fraction(lambda1)
    if lam.denominator!=1:
        return (NON_DET, {"raison":"e1 != 1 : construction generale non demontree"})
    det={}
    det["degre"]=poly_deg(phi2); det["monique"]=(phi2[-1]==1)
    if not det["monique"] or det["degre"]<1:
        return (ECHEC, det)
    exp=expansion_phi_adique(phi2,phi1)
    det["expansion_phi1"]=exp
    if not verifier_expansion(phi2,phi1,exp):
        return (ECHEC, det)
    pts=[(i,v0_poly(a)) for i,a in enumerate(exp) if any(a)]
    hull=polygone_newton(pts)
    det["points_phi2"]=pts; det["hull_phi2"]=hull
    if len(hull)!=2:
        return (ECHEC, det)
    (x0,y0),(x1,y1)=hull[0],hull[1]
    pente=Fraction(-(y0-y1),x1-x0); det["pente_phi2"]=pente
    if pente!=Fraction(-Fraction(lambda1)):
        return (ECHEC, det)
    segs=residuel_ordre1(phi1,phi2)
    det["segments_phi2"]=segs
    if len(segs)!=1:
        return (ECHEC, det)
    _,_,_,res_phi2=segs[0]
    det["residuel_phi2"]=res_phi2; det["psi1_attendu"]=psi1
    if not (isinstance(res_phi2,int) and isinstance(psi1,int)):
        return (NON_DET, det)
    if res_phi2!=psi1:
        return (ECHEC, det)
    if ((psi1>>degF(psi1))&1)!=1:
        return (ECHEC, det)
    h1=lam.numerator; e1=lam.denominator
    diff=[phi2[i]-(phi1[i] if i<len(phi1) else 0) for i in range(max(len(phi2),len(phi1)))]
    while len(diff)>1 and diff[-1]==0: diff.pop()
    det["diff_phi2_phi1"]=diff
    if poly_deg(diff)!=0:
        return (ECHEC, det)
    dval=diff[0]
    if v2_int(dval)!=h1:
        return (ECHEC, det)
    if abs(dval)//(2**h1)%2!=1:
        return (ECHEC, det)
    deg_attendu=poly_deg(phi1)*e1*degF(psi1)
    det["deg_attendu"]=deg_attendu
    if det["degre"]!=deg_attendu:
        return (ECHEC, det)
    return (OK, det)

def taylor_voie_B(T,phi):
    assert poly_deg(phi)==1 and phi[1]==1
    a0=-phi[0]
    b=[]
    for j in range(len(T)):
        s=0
        for i in range(j,len(T)):
            s+=T[i]*math.comb(i,j)*(a0**(i-j))
        b.append(s)
    while len(b)>1 and b[-1]==0: b.pop()
    return b

def verifier_branches_completes(branches_observes, couples_attendus):
    if couples_attendus is None:
        return (NON_DET, None, "Couples attendus archives non disponibles.")
    for (p_att, r_att) in couples_attendus:
        trouve=False
        for q in branches_observes:
            if (p_att, r_att) in branches_observes[q]:
                trouve=True; break
        if not trouve:
            return (ECHEC, False, "Couple attendu absent.")
    return (OK, True, "Tous couples attendus presents.")
