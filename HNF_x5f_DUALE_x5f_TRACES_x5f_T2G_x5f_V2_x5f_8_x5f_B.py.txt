def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def sha256_text(text):
    return hashlib.sha256(text.encode("utf-8")).hexdigest()

def validate_matrix(matrix):
    if not isinstance(matrix, list) or not matrix:
        raise ValueError("Matrice non vide requise")
    n = len(matrix)
    if any(not isinstance(row, list) or len(row) != n for row in matrix):
        raise ValueError("Matrice carree requise")
    for row in matrix:
        for value in row:
            if type(value) is not int:
                raise TypeError("Chaque coefficient doit etre un entier Python strict")
    return n


def identity(n):
    return [[int(i == j) for j in range(n)] for i in range(n)]


def matmul(a, b):
    if not a or not b:
        raise ValueError("Matrices non vides requises")
    if len(a[0]) != len(b):
        raise ValueError("Dimensions incompatibles")
    return [[sum(a[i][k] * b[k][j] for k in range(len(b)))
             for j in range(len(b[0]))] for i in range(len(a))]


def egcd(a, b):
    old_r, r = a, b
    old_s, s = 1, 0
    old_t, t = 0, 1
    while r:
        q = old_r // r
        old_r, r = r, old_r - q * r
        old_s, s = s, old_s - q * s
        old_t, t = t, old_t - q * t
    if old_r < 0:
        old_r = -old_r
        old_s = -old_s
        old_t = -old_t
    return old_r, old_s, old_t


def determinant_bareiss(matrix):
    # Utilise uniquement pour les petits cas / tests synthetiques.
    n = validate_matrix(matrix)
    a = [row[:] for row in matrix]
    if n == 1:
        return a[0][0]
    sign = 1
    previous = 1
    for k in range(n - 1):
        pivot_row = None
        for i in range(k, n):
            if a[i][k] != 0:
                pivot_row = i
                break
        if pivot_row is None:
            return 0
        if pivot_row != k:
            a[k], a[pivot_row] = a[pivot_row], a[k]
            sign = -sign
        pivot = a[k][k]
        for i in range(k + 1, n):
            for j in range(k + 1, n):
                numerator = a[i][j] * pivot - a[i][k] * a[k][j]
                if numerator % previous != 0:
                    raise ArithmeticError("Division Bareiss non exacte")
                a[i][j] = numerator // previous
            a[i][k] = 0
        previous = pivot
    return sign * a[n - 1][n - 1]

def parse_H_from_V10(text):
    """Parse la ligne H=... de l'archive V10. Retourne dict degre -> coeff."""
    hline = None
    for line in text.splitlines():
        if line.startswith("H="):
            hline = line[2:]
            break
    if hline is None:
        raise ValueError("Ligne H= introuvable dans V10")
    parts = re.split(r"\s\+\s|\s-\s", hline)
    signs = re.findall(r"\s([+-])\s", hline)
    if len(parts) != 40 or len(signs) != 39:
        raise ValueError("H doit avoir 40 termes (degres 0..39)")
    H = {}
    for idx, part in enumerate(parts):
        sgn = 1 if idx == 0 else (1 if signs[idx - 1] == "+" else -1)
        part = part.strip()
        m = re.match(r"(\d+)\*r\*\*(\d+)$", part)
        if m:
            H[int(m.group(2))] = sgn * int(m.group(1))
            continue
        m = re.match(r"(\d+)\*r$", part)
        if m:
            H[1] = sgn * int(m.group(1))
            continue
        m = re.match(r"(\d+)$", part)
        if m:
            H[0] = sgn * int(m.group(1))
            continue
        raise ValueError("Terme H non parse : %r" % part[:100])
    if sorted(H.keys()) != list(range(40)):
        raise ValueError("Degres H incomplets")
    return H


def reconstruct_T(H):
    """T(x) = sum c_i x^i, c_39 = 1, c_i = H_i * lc^(38-i) pour i < 39.
    Relation : H(y) = T(lc*y) / lc^38, avec lc = H_39."""
    lc = H[39]
    if lc <= 0:
        raise ValueError("lc = H_39 doit etre > 0")
    c = [0] * 40
    for i in range(39):
        c[i] = H[i] * (lc ** (38 - i))
    c[39] = 1
    return c


def canonical_sha_T(c):
    """Serialisation canonique : coefficients en ordre decroissant,
    un par ligne, avec \\n final."""
    ser = "\n".join(str(c[i]) for i in range(39, -1, -1)) + "\n"
    return sha256_text(ser)


def newton_sums(a, K):
    """Sommes de Newton s_0..s_K pour T monique a[0..n], a[n]=1.
    k <= n : s_k + sum_{i=1}^{k-1} a_{n-i} s_{k-i} + k*a_{n-k} = 0.
    k >  n : s_k + sum_{i=1}^{n} a_{n-i} s_{k-i} = 0.
    (La somme va jusqu'a k-1, pas k : corrige et verifie sur 4 polynomes tests.)"""
    n = len(a) - 1
    s = [0] * (K + 1)
    s[0] = n
    for k in range(1, K + 1):
        if k <= n:
            total = k * a[n - k]
            for i in range(1, k):
                total += a[n - i] * s[k - i]
            s[k] = -total
        else:
            total = 0
            for i in range(1, n + 1):
                total += a[n - i] * s[k - i]
            s[k] = -total
    return s


def det_mod(M, p):
    """Determinant modulo p par elimination de Gauss (independant de Bareiss/HNF)."""
    n = len(M)
    A = [[x % p for x in row] for row in M]
    det = 1
    for k in range(n):
        piv = None
        for i in range(k, n):
            if A[i][k] % p != 0:
                piv = i
                break
        if piv is None:
            return 0
        if piv != k:
            A[k], A[piv] = A[piv], A[k]
            det = -det
        pv = A[k][k] % p
        det = det * pv % p
        inv = pow(pv, -1, p)
        for i in range(k + 1, n):
            f = A[i][k] * inv % p
            if f:
                for j in range(k, n):
                    A[i][j] = (A[i][j] - f * A[k][j]) % p
    return det % p

def phase_reconstruct(state):
    t0 = time.time()
    # Authentification des sources
    sha_v10 = sha256_file(SRC_V10)
    sha_disc = sha256_file(SRC_DISC)
    if sha_v10 != SHA_V10:
        raise ValueError("SHA V10 mismatch : %s" % sha_v10)
    if sha_disc != SHA_DISC:
        raise ValueError("SHA discriminant mismatch : %s" % sha_disc)
    with open(SRC_V10, "r", encoding="utf-8") as f:
        v10_text = f.read()
    H = parse_H_from_V10(v10_text)
    lc = H[39]
    c = reconstruct_T(H)
    # Controles T
    assert len(c) == 40 and c[39] == 1, "T doit etre monique degre 39"
    assert all(type(x) is int for x in c), "Coefficients T entiers"
    sha_t = canonical_sha_T(c)
    if sha_t != SHA_T_CANON:
        raise ValueError("SHA canonique T mismatch : %s" % sha_t)
    # Comparaison independante avec COEFFICIENTS_T_BETA_V3.txt
    coeff_match = None
    if os.path.isfile(SRC_COEFF):
        coeffs = {}
        with open(SRC_COEFF, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line.startswith("c_"):
                    k, v = line.split("=", 1)
                    coeffs[int(k[2:])] = int(v)
        coeff_match = all(coeffs.get(i) == c[i] for i in range(40))
        if not coeff_match:
            raise ValueError("T reconstruit != COEFFICIENTS_T_BETA_V3.txt")
    # Newton
    t_newton = time.time()
    s = newton_sums(c, 76)
    assert len(s) == 77 and s[0] == 39
    # s_1 doit valoir -c_38 (somme des racines)
    assert s[1] == -c[38], "s_1 != -c_38"
    t_newton_dur = time.time() - t_newton
    # Matrice G
    t_g = time.time()
    G = [[s[i + j] for j in range(N)] for i in range(N)]
    assert all(G[i][j] == G[j][i] for i in range(N) for j in range(N)), "G non symetrique"
    t_g_dur = time.time() - t_g
    # Discriminant archive
    with open(SRC_DISC, "r", encoding="utf-8") as f:
        D = int(f.read().strip())
    # Ecriture des livrables T / Newton / G
    with open(OUT_T, "w", encoding="utf-8") as f:
        f.write("# T_BETA_COEFFICIENTS_V2_8_B.txt\n")
        f.write("# Polynome monique T de degre 39, coefficients c_0..c_39 (ordre croissant).\n")
        f.write("# Reconstruction : c_i = H_i * lc^(38-i) pour i<39, c_39=1, depuis %s (ligne H=).\n" % SRC_V10)
        f.write("# lc = H_39 = %d\n" % lc)
        f.write("# SHA canonique (ordre decroissant, un coeff par ligne, \\n final) : %s\n" % sha_t)
        for i in range(40):
            f.write("c_%d=%d\n" % (i, c[i]))
    with open(OUT_NEWTON, "w", encoding="utf-8") as f:
        f.write("# NEWTON_SUMS_T_BETA_S0_S76_V2_8_B.txt\n")
        f.write("# Sommes de Newton s_k = sum beta_i^k, k=0..76, pour T.\n")
        for k in range(77):
            f.write("s_%d=%d\n" % (k, s[k]))
    with open(OUT_G, "w", encoding="utf-8") as f:
        f.write("# MATRICE_TRACES_G_T2G_39_V2_8_B.txt\n")
        f.write("# Matrice de traces G, 39x39, G_ij = s_{i+j}, ordre ligne par ligne.\n")
        f.write("# Format : un entier par ligne, 1521 lignes (i=0..38, j=0..38).\n")
        for i in range(N):
            for j in range(N):
                f.write("%d\n" % G[i][j])
    save_pickle("T.pkl", {"c": c, "lc": lc, "sha_t": sha_t, "coeff_match": coeff_match})
    save_pickle("Newton.pkl", {"s": s})
    save_pickle("G.pkl", {"G": G})
    save_pickle("Disc.pkl", {"D": D})
    dur = time.time() - t0
    state["phases"]["reconstruct"] = "PASS"
    state["timings"]["reconstruct_total_s"] = round(dur, 1)
    state["timings"]["newton_s"] = round(t_newton_dur, 1)
    state["timings"]["build_G_s"] = round(t_g_dur, 1)
    state["reconstruct"] = {
        "sha_v10": sha_v10, "sha_disc": sha_disc, "sha_T_canon": sha_t,
        "lc_digits": len(str(lc)), "c0_digits": len(str(abs(c[0]))),
        "coeff_v3_match": coeff_match,
        "s76_digits": len(str(abs(s[76]))),
        "G_max_digits": max(len(str(abs(v))) for row in G for v in row),
        "D_digits": len(str(abs(D))),
    }
    save_state(state)
    print("RECONSTRUCT PASS (%.1fs) : T SHA=%s, coeff V3 match=%s, s76 chiffres=%d, G max=%d chiffres"
          % (dur, sha_t, coeff_match, len(str(abs(s[76]))),
             max(len(str(abs(v))) for row in G for v in row)))
    return state

# 4. PHASE HNF (moteur V2.8-A corrige + comptage det par construction)

def phase_hnf(state, steps):
    t0 = time.time()
    if has_pickle("HNF.pkl"):
        st = load_pickle("HNF.pkl")
        h, u, k_start = st["h"], st["u"], st["k"]
        swaps, negs, combines = st["swaps"], st["negs"], st["combines"]
    else:
        G = load_pickle("G.pkl")["G"]
        h = [row[:] for row in G]
        u = identity(N)
        k_start = 0
        swaps = negs = combines = 0
    k_end = min(N, k_start + steps)

    def combine_rows(i, j, a, b, cc, d):
        nonlocal combines
        combines += 1
        hi = h[i][:]; hj = h[j][:]
        ui = u[i][:]; uj = u[j][:]
        h[i] = [a * x + b * y for x, y in zip(hi, hj)]
        h[j] = [cc * x + d * y for x, y in zip(hi, hj)]
        u[i] = [a * x + b * y for x, y in zip(ui, uj)]
        u[j] = [cc * x + d * y for x, y in zip(ui, uj)]

    for k in range(k_start, k_end):
        pr = None
        for i in range(k, N):
            if h[i][k] != 0:
                pr = i
                break
        if pr is None:
            raise ValueError("Matrice singuliere (HNF k=%d)" % k)
        if pr != k:
            h[k], h[pr] = h[pr], h[k]
            u[k], u[pr] = u[pr], u[k]
            swaps += 1
        for i in range(k + 1, N):
            b = h[i][k]
            if b == 0:
                continue
            a = h[k][k]
            g, x, y = egcd(a, b)
            if g == 0:
                raise ArithmeticError("PGCD nul inattendu")
            # det(combine) = (x*a + y*b)/g = 1 : verifie a chaque fois
            assert x * a + y * b == g
            combine_rows(k, i, x, y, -b // g, a // g)
        if h[k][k] < 0:
            h[k] = [-v for v in h[k]]
            u[k] = [-v for v in u[k]]
            negs += 1
        piv = h[k][k]
        if piv <= 0:
            raise ArithmeticError("Pivot HNF non positif")
        for i in range(k):
            q = h[i][k] // piv
            if q:
                # Transvection : det 1
                h[i] = [x - q * y for x, y in zip(h[i], h[k])]
                u[i] = [x - q * y for x, y in zip(u[i], u[k])]
        print("  HNF k=%d : pivot chiffres=%d" % (k, len(str(abs(piv)))), flush=True)
    save_pickle("HNF.pkl", {"h": h, "u": u, "k": k_end,
                            "swaps": swaps, "negs": negs, "combines": combines})
    dur = time.time() - t0
    state["timings"]["hnf_cumul_s"] = state["timings"].get("hnf_cumul_s", 0) + round(dur, 1)
    if k_end == N:
        # Controles de structure
        tri = all(h[i][j] == 0 for i in range(N) for j in range(i))
        diag = all(h[i][i] > 0 for i in range(N))
        red = all(0 <= h[i][j] < h[j][j] for i in range(N) for j in range(i + 1, N))
        if not (tri and diag and red):
            raise AssertionError("Structure HNF invalide")
        det_u = (-1) ** (swaps + negs)  # par construction : combine/transvection det 1
        detH = 1
        for i in range(N):
            detH *= h[i][i]
        save_pickle("detH.pkl", {"detH": detH, "detU": det_u})
        with open(OUT_H, "w", encoding="utf-8") as f:
            f.write("# HNF_G_T2G_39_V2_8_B.txt\n")
            f.write("# HNF H, 39x39, triangulaire superieure, diagonale > 0, reduite.\n")
            f.write("# Format : un entier par ligne, 1521 lignes.\n")
            for i in range(N):
                for j in range(N):
                    f.write("%d\n" % h[i][j])
        with open(OUT_U, "w", encoding="utf-8") as f:
            f.write("# TRANSFORMATION_U_T2G_39_V2_8_B.txt\n")
            f.write("# Transformation unimodulaire U, 39x39, H = U*G, det(U) = %d (par construction).\n" % det_u)
            f.write("# Format : un entier par ligne, 1521 lignes.\n")
            for i in range(N):
                for j in range(N):
                    f.write("%d\n" % u[i][j])
        state["phases"]["hnf"] = "PASS"
        state["hnf"] = {"swaps": swaps, "negs": negs, "combines": combines,
                        "detU_by_construction": det_u,
                        "detH_digits": len(str(detH)),
                        "H_max_digits": max(len(str(abs(v))) for row in h for v in row),
                        "U_max_digits": max(len(str(abs(v))) for row in u for v in row)}
        print("HNF PASS : swaps=%d negs=%d combines=%d det(U)=%d det(H) chiffres=%d"
              % (swaps, negs, combines, det_u, len(str(detH))))
    else:
        state["phases"]["hnf"] = "PARTIEL k=%d/39" % k_end
        print("HNF PARTIEL : k=%d/39" % k_end)
    save_state(state)
    return state

def phase_adj(state, steps):
    t0 = time.time()
    h = load_pickle("HNF.pkl")["h"]
    detH = load_pickle("detH.pkl")["detH"]
    if has_pickle("Adj.pkl"):
        st = load_pickle("Adj.pkl")
        Y, j_start = st["Y"], st["j"]
    else:
        Y = [[0] * N for _ in range(N)]
        j_start = 0
    j_end = min(N, j_start + steps)
    for j in range(j_start, j_end):
        assert detH % h[j][j] == 0
        Y[j][j] = detH // h[j][j]
        for i in range(j - 1, -1, -1):
            total = 0
            for k in range(i + 1, j + 1):
                total += h[i][k] * Y[k][j]
            assert total % h[i][i] == 0, "Adjointe : division non exacte"
            Y[i][j] = -total // h[i][i]
        print("  ADJ colonne j=%d" % j, flush=True)
    save_pickle("Adj.pkl", {"Y": Y, "j": j_end})
    dur = time.time() - t0
    state["timings"]["adj_cumul_s"] = state["timings"].get("adj_cumul_s", 0) + round(dur, 1)
    state["phases"]["adj"] = "PASS" if j_end == N else "PARTIEL j=%d/39" % j_end
    save_state(state)
    print("ADJ %s : j=%d/39" % (state["phases"]["adj"], j_end))
    return state

# 6. PHASE HINV (H^(-1) en fractions reduites, par substitution)
# ============================================================

def phase_hinv(state, steps):
    t0 = time.time()
    h = load_pickle("HNF.pkl")["h"]
    if has_pickle("Hinv.pkl"):
        st = load_pickle("Hinv.pkl")
        X, j_start = st["X"], st["j"]
    else:
        X = [[Fraction(0)] * N for _ in range(N)]
        j_start = 0
    j_end = min(N, j_start + steps)
    for j in range(j_start, j_end):
        X[j][j] = Fraction(1, h[j][j])
        for i in range(j - 1, -1, -1):
            total = Fraction(0)
            for k in range(i + 1, j + 1):
                if h[i][k] != 0 and X[k][j] != 0:
                    total += h[i][k] * X[k][j]
            X[i][j] = -total / h[i][i]
        print("  HINV colonne j=%d" % j, flush=True)
    save_pickle("Hinv.pkl", {"X": X, "j": j_end})
    dur = time.time() - t0
    state["timings"]["hinv_cumul_s"] = state["timings"].get("hinv_cumul_s", 0) + round(dur, 1)
    if j_end == N:
        with open(OUT_DUAL, "w", encoding="utf-8") as f:
            f.write("# BASE_DUALE_TRACES_T2G_39_V2_8_B.txt\n")
            f.write("# Base duale : colonnes de H^(-1), fractions exactes reduites.\n")
            f.write("# Format : une fraction 'numerateur/denominateur' par ligne, 1521 lignes\n")
            f.write("# (i=0..38, j=0..38). Les entrees sous la diagonale sont 0/1.\n")
            f.write("# Verification : G * H^(-1) = U^(-1) (entier), voir rapport.\n")
            for i in range(N):
                for j in range(N):
                    v = X[i][j]
                    f.write("%d/%d\n" % (v.numerator, v.denominator))
        state["phases"]["hinv"] = "PASS"
        state["hinv"] = {
            "max_num_digits": max(len(str(abs(X[i][j].numerator))) for i in range(N) for j in range(N)),
            "max_den_digits": max(len(str(abs(X[i][j].denominator))) for i in range(N) for j in range(N)),
        }
        print("HINV PASS : num max=%d chiffres, den max=%d chiffres"
              % (state["hinv"]["max_num_digits"], state["hinv"]["max_den_digits"]))
    else:
        state["phases"]["hinv"] = "PARTIEL j=%d/39" % j_end
        print("HINV PARTIEL : j=%d/39" % j_end)
    save_state(state)
    return state

# 7. PHASE DUAL (W = G*Adj(H), Z = W/det(H) = U^(-1))
# ============================================================

def phase_dual(state, rows):
    t0 = time.time()
    G = load_pickle("G.pkl")["G"]
    Y = load_pickle("Adj.pkl")["Y"]
    detH = load_pickle("detH.pkl")["detH"]
    if has_pickle("W.pkl"):
        st = load_pickle("W.pkl")
        W, i_start = st["W"], st["i"]
    else:
        W = [[0] * N for _ in range(N)]
        i_start = 0
    i_end = min(N, i_start + rows)
    for i in range(i_start, i_end):
        for j in range(N):
            total = 0
            for k in range(N):
                total += G[i][k] * Y[k][j]
            W[i][j] = total
        print("  DUAL ligne W i=%d" % i, flush=True)
    save_pickle("W.pkl", {"W": W, "i": i_end})
    dur = time.time() - t0
    state["timings"]["dual_cumul_s"] = state["timings"].get("dual_cumul_s", 0) + round(dur, 1)
    if i_end == N:
        nondiv = 0
        Z = [[0] * N for _ in range(N)]
        for i in range(N):
            for j in range(N):
                if W[i][j] % detH != 0:
                    nondiv += 1
                else:
                    Z[i][j] = W[i][j] // detH
        if nondiv != 0:
            raise AssertionError("G*Adj(H) non divisible par det(H) : %d entrees" % nondiv)
        save_pickle("Z.pkl", {"Z": Z})
        state["phases"]["dual"] = "PASS"
        state["dual"] = {"nondiv": nondiv,
                         "Z_max_digits": max(len(str(abs(v))) for row in Z for v in row)}
        print("DUAL PASS : Z entier, 1521/1521 divisibles, Z max=%d chiffres"
              % state["dual"]["Z_max_digits"])
    else:
        state["phases"]["dual"] = "PARTIEL i=%d/39" % i_end
        print("DUAL PARTIEL : i=%d/39" % i_end)
    save_state(state)
    return state

# 8. PHASE VERIFY (verifications exactes finales)
# ============================================================

def phase_verify(state):
    t0 = time.time()
    G = load_pickle("G.pkl")["G"]
    h = load_pickle("HNF.pkl")["h"]
    u = load_pickle("HNF.pkl")["u"]
    detH = load_pickle("detH.pkl")["detH"]
    detU = load_pickle("detH.pkl")["detU"]
    Y = load_pickle("Adj.pkl")["Y"]
    X = load_pickle("Hinv.pkl")["X"]
    Z = load_pickle("Z.pkl")["Z"]
    D = load_pickle("Disc.pkl")["D"]
    checks = {}
    # H = U*G (ligne par ligne)
    checks["H_equals_U_times_G"] = (matmul(u, G) == h)
    # det(H) = D (exact)
    checks["det_H_equals_disc"] = (detH == D)
    checks["det_G_equals_disc_via_HNF"] = checks["det_H_equals_disc"] and (detU == 1)
    # det(G) mod p independant (Gauss mod p)
    mod_primes = [101, 1009, 10007, 99991, 1000003]
    mod_ok = all(det_mod(G, p) == D % p for p in mod_primes)
    checks["det_mod_primes"] = mod_ok
    checks["mod_primes"] = mod_primes
    # H * Adj(H) == det * I.
    # Adj(H) est CONSTRUIT en resolvant H*Y = det*I par substitution arriere
    # (phase adj) : l'identite tient par construction. On la re-verifie ici
    # par sondage sur 5 entrees (diagonale + hors-diagonale), car la
    # re-verification complete (1521 produits scalaires d'entiers de 164k chiffres)
    # est redondante avec la construction et couterait ~90 s.
    adj_ok = True
    for (i, j) in [(0, 0), (1, 1), (38, 38), (0, 1), (0, 38)]:
        tot = sum(h[i][k] * Y[k][j] for k in range(N))
        if tot != (detH if i == j else 0):
            adj_ok = False
            break
    checks["H_times_Adj_equals_det_I"] = adj_ok
    checks["H_times_Adj_note"] = "Adj construit en resolvant H*Y=det*I ; re-verifie par sondage sur 5 entrees."
    # H * H^(-1) == I (spot checks + diagonale complete via fractions)
    hinv_ok = True
    for i in range(N):
        tot = sum(Fraction(h[i][k]) * X[k][i] for k in range(N))
        if tot != 1:
            hinv_ok = False
            break
    checks["H_times_Hinv_diag_I"] = hinv_ok
    # G * H^(-1) == Z (Fractions, complet)
    ghinv_ok = True
    for j in range(N):
        for i in range(N):
            tot = Fraction(0)
            for k in range(N):
                if X[k][j] != 0:
                    tot += G[i][k] * X[k][j]
            if tot.denominator != 1 or int(tot) != Z[i][j]:
                ghinv_ok = False
                break
        if not ghinv_ok:
            break
    checks["G_times_Hinv_equals_Z"] = ghinv_ok
    # Z*U == I et U*Z == I (bilateral)
    checks["Z_times_U_equals_I"] = (matmul(Z, u) == identity(N))
    checks["U_times_Z_equals_I"] = (matmul(u, Z) == identity(N))
    # det(Z) = 1 (car Z*U=I, entiers)
    checks["det_Z_equals_1_via_bilateral"] = checks["Z_times_U_equals_I"]
    # Structure HNF (re-verifiee)
    checks["H_upper_triangular"] = all(h[i][j] == 0 for i in range(N) for j in range(i))
    checks["H_positive_diagonal"] = all(h[i][i] > 0 for i in range(N))
    checks["H_reduced"] = all(0 <= h[i][j] < h[j][j] for i in range(N) for j in range(i + 1, N))
    checks["Hinv_rational_exact"] = all(isinstance(X[i][j], Fraction) for i in range(N) for j in range(N))
    dur = time.time() - t0
    state["timings"]["verify_s"] = round(dur, 1)
    state["verify"] = checks
    state["phases"]["verify"] = "PASS" if all(v for k, v in checks.items() if isinstance(v, bool)) else "FAIL"
    save_state(state)
    print("VERIFY %s (%.1fs) : %s" % (state["phases"]["verify"], dur,
                                      {k: v for k, v in checks.items() if isinstance(v, bool)}))
    if state["phases"]["verify"] != "PASS":
        raise AssertionError("Echec verification : %s" % checks)
    return state

# 9. TESTS SYNTHETIQUES (+ nouveaux tests V2.8-B)
# ============================================================

def phase_selftest(state):
    t0 = time.time()
    results = {}
    # --- Moteur V2.8-A : 3 references + aleatoires (version reduite rapide) ---
    # On reutilise les fonctions locales (meme algorithme que V2.8-A corrige).
    def row_hnf_with_transform(matrix):
        n = validate_matrix(matrix)
        h = [row[:] for row in matrix]
        u = identity(n)

        def combine(i, j, a, b, cc, d):
            hi = h[i][:]; hj = h[j][:]
            ui = u[i][:]; uj = u[j][:]
            h[i] = [a * x + b * y for x, y in zip(hi, hj)]
            h[j] = [cc * x + d * y for x, y in zip(hi, hj)]
            u[i] = [a * x + b * y for x, y in zip(ui, uj)]
            u[j] = [cc * x + d * y for x, y in zip(ui, uj)]
        for k in range(n):
            pr = None
            for i in range(k, n):
                if h[i][k] != 0:
                    pr = i
                    break
            if pr is None:
                raise ValueError("Matrice singuliere")
            if pr != k:
                h[k], h[pr] = h[pr], h[k]
                u[k], u[pr] = u[pr], u[k]
            for i in range(k + 1, n):
                b = h[i][k]
                if b == 0:
                    continue
                a = h[k][k]
                g, x, y = egcd(a, b)
                combine(k, i, x, y, -b // g, a // g)
            if h[k][k] < 0:
                h[k] = [-v for v in h[k]]
                u[k] = [-v for v in u[k]]
            piv = h[k][k]
            for i in range(k):
                q = h[i][k] // piv
                if q:
                    h[i] = [x - q * y for x, y in zip(h[i], h[k])]
                    u[i] = [x - q * y for x, y in zip(u[i], u[k])]
        return h, u

    def verify_small(g, h, u):
        n = validate_matrix(g)
        assert matmul(u, g) == h
        assert abs(determinant_bareiss(u)) == 1
        for i in range(n):
            assert all(h[i][j] == 0 for j in range(i))
            assert h[i][i] > 0
            assert all(0 <= h[i][j] < h[j][j] for j in range(i + 1, n))
        dg = determinant_bareiss(g)
        dh = determinant_bareiss(h)
        assert abs(dg) == dh, "det(H) != |det(G)| (correction V2.8-A)"
        return True

    refs = [[[5, -2, 1], [-2, 4, -2], [1, -2, 5]],
            [[2, 1, 0, 0], [1, 3, 1, 0], [0, 1, 4, 1], [0, 0, 1, 5]],
            [[2, 1, 3], [1, 2, 0], [0, 1, 2]]]
    for idx, g in enumerate(refs, 1):
        h, u = row_hnf_with_transform(g)
        verify_small(g, h, u)
    results["reference_cases"] = 3
    rng = random.Random(20261124)
    cnt = 0
    for n in range(1, 8):
        acc = 0
        while acc < 100:
            g = [[rng.randint(-20, 20) for _ in range(n)] for __ in range(n)]
            if determinant_bareiss(g) == 0:
                continue
            h, u = row_hnf_with_transform(g)
            verify_small(g, h, u)
            acc += 1
            cnt += 1
    results["random_matrices_verified"] = cnt
    # --- Nouveaux tests V2.8-B ---
    # G = [[-9]] (cas qui a fait echouer V2.8-A avant correction)
    h, u = row_hnf_with_transform([[-9]])
    assert h == [[9]] and u == [[-1]]
    verify_small([[-9]], h, u)
    results["G_minus9"] = True
    # Determinant negatif en dimension 2
    h, u = row_hnf_with_transform([[2, 1], [3, 4]])
    assert determinant_bareiss([[2, 1], [3, 4]]) == 5  # det>0 ici ; autre cas :
    h2, u2 = row_hnf_with_transform([[0, 1], [1, 0]])
    assert determinant_bareiss([[0, 1], [1, 0]]) == -1
    verify_small([[0, 1], [1, 0]], h2, u2)
    results["det_negative_dim2"] = True
    # Singularite
    try:
        row_hnf_with_transform([[1, 2], [2, 4]])
        results["singular_rejected"] = False
    except ValueError:
        results["singular_rejected"] = True
    # Corruption de H et U (doivent etre detectees)
    g = [[5, -2, 1], [-2, 4, -2], [1, -2, 5]]
    h, u = row_hnf_with_transform(g)
    h_bad = [row[:] for row in h]
    h_bad[0][0] += 1
    try:
        verify_small(g, h_bad, u)
        results["corrupt_H_detected"] = False
    except AssertionError:
        results["corrupt_H_detected"] = True
    u_bad = [row[:] for row in u]
    u_bad[0][0] += 1
    try:
        verify_small(g, h, u_bad)
        results["corrupt_U_detected"] = False
    except AssertionError:
        results["corrupt_U_detected"] = True
    # Hash des sources
    results["hash_V10_ok"] = (sha256_file(SRC_V10) == SHA_V10)
    results["hash_disc_ok"] = (sha256_file(SRC_DISC) == SHA_DISC)
    # Newton sur polynome de reference simple x^2-3x+2 (racines 1,2)
    s_ref = newton_sums([2, -3, 1], 5)
    results["newton_reference"] = (s_ref == [2, 3, 5, 9, 17, 33])
    # Newton sur x^3-6x^2+11x-6 (racines 1,2,3)
    results["newton_reference2"] = (newton_sums([-6, 11, -6, 1], 4) == [3, 6, 14, 36, 98])
    dur = time.time() - t0
    state["selftest"] = results
    state["timings"]["selftest_s"] = round(dur, 1)
    state["phases"]["selftest"] = "PASS" if all(results.values()) else "FAIL"
    save_state(state)
    print("SELFTEST %s (%.1fs) : %s" % (state["phases"]["selftest"], dur, results))
    if state["phases"]["selftest"] != "PASS":
        raise AssertionError("Echec selftest")
    return state

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--selftest", action="store_true")
    parser.add_argument("--phase", choices=["reconstruct", "hnf", "adj", "hinv", "dual", "verify", "report", "all"], default=None)
    parser.add_argument("--steps", type=int, default=39)
    parser.add_argument("--rows", type=int, default=39)
    args = parser.parse_args()
    state = load_state()
    if args.selftest:
        phase_selftest(state)
        return
    phase = args.phase or "all"
    if phase == "all":
        phase_reconstruct(state)
        phase_hnf(state, 39)
        phase_adj(state, 39)
        phase_hinv(state, 39)
        phase_dual(state, 39)
        phase_verify(state)
        phase_selftest(state)
        phase_report(state)
        return
    if phase == "reconstruct":
        phase_reconstruct(state)
    elif phase == "hnf":
        phase_hnf(state, args.steps)
    elif phase == "adj":
        phase_adj(state, args.steps)
    elif phase == "hinv":
        phase_hinv(state, args.steps)
    elif phase == "dual":
        phase_dual(state, args.rows)
    elif phase == "verify":
        phase_verify(state)
    elif phase == "report":
        phase_report(state)


if __name__ == "__main__":
    main()
